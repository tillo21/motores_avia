💻 Código Backend - WebSocket Server (Node.js)
server.js 
// AVIA WebSocket Notification Server
// Node.js + Socket.IO + JWT Authentication

const express = require('express');
const http = require('http');
const socketIO = require('socket.io');
const jwt = require('jsonwebtoken');
const rateLimit = require('express-rate-limit');

const app = express();
const server = http.createServer(app);
const io = socketIO(server, {
    cors: {
        origin: "*",
        methods: ["GET", "POST"]
    }
});

const JWT_SECRET = 'avia-secret-key-2026';
const PORT = 3000;

// Middleware
app.use(express.json());

// Rate Limiting
const limiter = rateLimit({
    windowMs: 60 * 1000, // 1 minuto
    max: 100 // 100 requests por minuto
});
app.use(limiter);

// Storage
const connectedClients = new Map();
const notificationHistory = [];

// Autenticación JWT Middleware para WebSocket
io.use((socket, next) => {
    const token = socket.handshake.auth.token;
    
    if (!token) {
        return next(new Error('Authentication error: No token provided'));
    }
    
    try {
        const decoded = jwt.verify(token, JWT_SECRET);
        socket.userId = decoded.userId;
        socket.userName = decoded.userName;
        next();
    } catch (err) {
        next(new Error('Authentication error: Invalid token'));
    }
});

// WebSocket Event Handlers
io.on('connection', (socket) => {
    console.log(`✅ Cliente conectado: ${socket.userName} (${socket.userId})`);
    
    // Registrar cliente
    connectedClients.set(socket.id, {
        userId: socket.userId,
        userName: socket.userName,
        connectedAt: new Date()
    });
    
    // Emitir estado de conexión
    io.emit('client:count', connectedClients.size);
    
    // Enviar historial de notificaciones recientes
    socket.emit('notification:history', notificationHistory.slice(-20));
    
    // Handler: Cliente solicita notificaciones
    socket.on('notification:request', () => {
        socket.emit('notification:history', notificationHistory);
    });
    
    // Handler: Marcar notificación como leída
    socket.on('notification:read', (notificationId) => {
        const notification = notificationHistory.find(n => n.id === notificationId);
        if (notification) {
            notification.read = true;
            socket.emit('notification:updated', notification);
        }
    });
    
    // Handler: Desconexión
    socket.on('disconnect', () => {
        console.log(`❌ Cliente desconectado: ${socket.userName}`);
        connectedClients.delete(socket.id);
        io.emit('client:count', connectedClients.size);
    });
    
    // Handler: Ping para medir latencia
    socket.on('ping', (callback) => {
        callback();
    });
});

// REST API Endpoints

// POST /api/auth/login - Generar JWT
app.post('/api/auth/login', (req, res) => {
    const { username, password } = req.body;
    
    // Validación simple (en producción usar bcrypt + DB)
    if (!username || !password) {
        return res.status(400).json({ error: 'Username y password requeridos' });
    }
    
    const token = jwt.sign(
        { 
            userId: `user_${Date.now()}`,
            userName: username 
        },
        JWT_SECRET,
        { expiresIn: '24h' }
    );
    
    res.json({ 
        token,
        expiresIn: 86400 
    });
});

// POST /api/notifications - Enviar notificación
app.post('/api/notifications', (req, res) => {
    const { type, title, message, targetUserId } = req.body;
    
    const notification = {
        id: `notif_${Date.now()}_${Math.random().toString(36).substr(2, 9)}`,
        type: type || 'info',
        title: title || 'Nueva Notificación',
        message: message || '',
        timestamp: new Date().toISOString(),
        read: false
    };
    
    // Guardar en historial
    notificationHistory.push(notification);
    if (notificationHistory.length > 100) {
        notificationHistory.shift();
    }
    
    // Emitir notificación
    if (targetUserId && targetUserId !== 'all') {
        // Enviar a usuario específico
        const targetSocket = Array.from(connectedClients.entries())
            .find(([_, client]) => client.userId === targetUserId);
        
        if (targetSocket) {
            io.to(targetSocket[0]).emit('notification:new', notification);
        }
    } else {
        // Broadcast a todos
        io.emit('notification:new', notification);
    }
    
    res.json({ 
        success: true, 
        notification 
    });
});

// GET /api/notifications - Obtener historial
app.get('/api/notifications', (req, res) => {
    res.json({ 
        notifications: notificationHistory,
        total: notificationHistory.length 
    });
});

// GET /api/stats - Estadísticas del servidor
app.get('/api/stats', (req, res) => {
    res.json({
        connectedClients: connectedClients.size,
        totalNotifications: notificationHistory.length,
        uptime: process.uptime(),
        memoryUsage: process.memoryUsage()
    });
});

// Iniciar servidor
server.listen(PORT, () => {
    console.log(`🚀 AVIA WebSocket Server corriendo en http://localhost:${PORT}`);
    console.log(`🔌 WebSocket endpoint: ws://localhost:${PORT}`);
    console.log(`📡 API REST: http://localhost:${PORT}/api`);
});

// Graceful shutdown
process.on('SIGTERM', () => {
    console.log('🛑 SIGTERM recibido, cerrando servidor...');
    server.close(() => {
        console.log('✅ Servidor cerrado correctamente');
        process.exit(0);
    });
});
💻 Código Cliente - WebSocket Client (JavaScript)
client.js 
// AVIA WebSocket Client
// Socket.IO Client + JWT Authentication

import io from 'socket.io-client';

class NotificationClient {
    constructor() {
        this.socket = null;
        this.token = null;
        this.callbacks = {
            onConnect: () => {},
            onDisconnect: () => {},
            onNotification: () => {},
            onError: () => {}
        };
    }

    // Autenticación y conexión
    async connect(username, password) {
        try {
            // 1. Obtener JWT token
            const response = await fetch('http://localhost:3000/api/auth/login', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ username, password })
            });
            
            const { token } = await response.json();
            this.token = token;
            
            // 2. Conectar WebSocket con token
            this.socket = io('http://localhost:3000', {
                auth: { token }
            });
            
            // 3. Event listeners
            this.socket.on('connect', () => {
                console.log('✅ Conectado al servidor WebSocket');
                this.callbacks.onConnect();
            });
            
            this.socket.on('disconnect', () => {
                console.log('❌ Desconectado del servidor');
                this.callbacks.onDisconnect();
            });
            
            this.socket.on('notification:new', (notification) => {
                console.log('📬 Nueva notificación:', notification);
                this.callbacks.onNotification(notification);
                this.displayNotification(notification);
            });
            
            this.socket.on('notification:history', (notifications) => {
                console.log('📚 Historial recibido:', notifications.length);
            });
            
            this.socket.on('error', (error) => {
                console.error('❌ Error:', error);
                this.callbacks.onError(error);
            });
            
            return true;
        } catch (error) {
            console.error('Error al conectar:', error);
            return false;
        }
    }

    // Mostrar notificación en UI
    displayNotification(notification) {
        const icons = {
            info: 'ℹ️',
            success: '✅',
            warning: '⚠️',
            error: '❌',
            message: '💬'
        };
        
        const notifElement = document.createElement('div');
        notifElement.className = 'toast-notification';
        notifElement.innerHTML = \`
            
\${icons[notification.type] || '📢'}

            

                
\${notification.title}

                
\${notification.message}

            

            
        \`;
        
        document.body.appendChild(notifElement);
        
        // Auto-remove después de 5 segundos
        setTimeout(() => notifElement.remove(), 5000);
    }

    // Marcar como leída
    markAsRead(notificationId) {
        this.socket.emit('notification:read', notificationId);
    }

    // Desconectar
    disconnect() {
        if (this.socket) {
            this.socket.disconnect();
        }
    }

    // Medir latencia
    async ping() {
        return new Promise((resolve) => {
            const start = Date.now();
            this.socket.emit('ping', () => {
                const latency = Date.now() - start;
                resolve(latency);
            });
        });
    }
}

// Uso
const client = new NotificationClient();

client.callbacks.onNotification = (notification) => {
    console.log('Notificación recibida:', notification);
};

// Conectar
await client.connect('usuario@avia.com', 'password123');

// Medir latencia
const latency = await client.ping();
console.log(\`Latencia: \${latency}ms\`);
