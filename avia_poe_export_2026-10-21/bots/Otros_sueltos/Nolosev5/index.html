aviaPipeline.executeSyncStep = function(stepConfig, callbacks) {
  var self = this;
  var steps = stepConfig.parallelSteps || [];
  
  // ❌ ES6-FAIL: Promise.all no es ES5
  return Promise.all(
    steps.map(function(step) {
      return new Promise(function(resolve, reject) {
        step.execute(function(err, result) {
          if (err) reject(err);
          else resolve(result);
        });
      });
    })
  ).then(function(results) {
    if (callbacks.onSuccess) callbacks.onSuccess(results);
    self.broadcastSync('step_completed', {results: results});
  }).catch(function(error) {
    if (callbacks.onError) callbacks.onError(error);
    self.logError('executeSyncStep failed', error);
  });
};

aviaPipeline.executeSyncStep = function(stepConfig, callbacks) {
  var self = this;
  var steps = stepConfig.parallelSteps || [];
  var results = [];
  var errors = [];
  var completed = 0;
  var total = steps.length;

  /**
   * Función auxiliar: verifica si todos los pasos han completado
   * ES5-compliant: function declaration + var
   */
  function checkCompletion() {
    if (completed === total) {
      if (errors.length > 0) {
        if (callbacks.onError) {
          callbacks.onError(errors);
        }
        self.logError('executeSyncStep partial failure', errors);
      } else {
        if (callbacks.onSuccess) {
          callbacks.onSuccess(results);
        }
        self.broadcastSync('step_completed', {results: results});
      }
      
      // Registro en Ledger (LEDGER-FAIL corregido)
      self.ledger.register({
        action: 'pipeline_sync_step_completed',
        timestamp: new Date().toISOString(),
        results: results.length,
        errors: errors.length,
        hash: self.generateHash('executeSyncStep:' + new Date().getTime())
      });
    }
  }

  /**
   * Ejecución paralela ES5-compliant
   * Usa IIFE para capturar correctamente index y step
   */
  for (var i = 0; i < steps.length; i++) {
    (function(index, step) {
      step.execute(function(err, result) {
        if (err) {
          errors.push({
            step: index,
            error: err,
            timestamp: new Date().toISOString()
          });
        } else {
          results[index] = result;
        }
        completed++;
        checkCompletion();
      });
    })(i, steps[i]);
  }

  /**
   * Caso edge: array de pasos vacío
   */
  if (total === 0) {
    checkCompletion();
  }
};
