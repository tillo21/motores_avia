// Redis Cache Configuration const redis = require('redis'); const client = redis.createClient({ host: 'localhost', port: 6379, db: 0, retry_strategy: (options) => Math.min(options.attempt * 100, 3000) }); // Cache Middleware const cacheMiddleware = (duration) => { return async (req, res, next) => { const key = `cache:${req.originalUrl}`; const cached = await client.get(key); if (cached) { return res.json(JSON.parse(cached)); } res.sendResponse = res.json; res.json = (body) => { client.setex(key, duration, JSON.stringify(body)); res.sendResponse(body); }; next(); }; }; // Usage app.get('/api/projects', cacheMiddleware(300), getProjects);

📚 Endpoints Principales

POST /api/auth/register

Registrar nuevo usuario. Requiere: email, password, name

POST /api/auth/login

Autenticar usuario. Retorna JWT token

GET /api/projects

Listar proyectos del usuario autenticado. Soporta paginación y filtros

POST /api/projects

Crear nuevo proyecto. Requiere: name, template, configuration

GET /api/projects/:id

Obtener detalles de proyecto específico

PUT /api/projects/:id

Actualizar configuración de proyecto

DELETE /api/projects/:id

Eliminar proyecto permanentemente

POST /api/projects/:id/deploy

Desplegar proyecto a producción

GET /api/deployments/:id/status

Verificar estado de despliegue en progreso

GET /api/analytics/dashboard

Métricas agregadas del dashboard principal
