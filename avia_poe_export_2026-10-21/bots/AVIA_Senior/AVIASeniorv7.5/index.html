async function generateAIResponse(message) {
    const POE_API_KEY = 'tu-api-key-aqui';
    const response = await fetch('https://api.poe.com/v1/chat', {
        method: 'POST',
        headers: {
            'Authorization': `Bearer ${POE_API_KEY}`,
            'Content-Type': 'application/json'
        },
        body: JSON.stringify({
            query: message,
            bot: 'Claude-3-Sonnet'
        })
    });
    return await response.json();
}
