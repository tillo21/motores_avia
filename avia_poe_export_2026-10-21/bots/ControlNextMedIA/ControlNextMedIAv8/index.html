Instalar Certbot en container Nginx docker compose exec nginx apk add --no-cache certbot certbot-nginx # 2. Emitir certificado wildcard docker compose exec nginx certbot certonly \ --nginx \ --non-interactive \ --agree-tos \ --email $EMAIL \ -d $DOMAIN \ -d www.$DOMAIN \ -d api.$DOMAIN # 3. Recargar Nginx docker compose exec nginx nginx -s reload echo "✅ SSL configurado para $DOMAIN" # 4. Cron: auto-renovación cada 60 días echo "0 3 */60 * * certbot renew --quiet && nginx -s reload" \ | crontab - # 5. Test HTTPS curl -I https://$DOMAIN/health curl -I https://$DOMAIN/api/v1/health
