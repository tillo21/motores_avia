// ── useSocket · T3.7 · ControlNextMedIA ── import { useEffect, useRef, useCallback, useState } from 'react'; import { io, Socket } from 'socket.io-client'; import { BotRouter } from './BotRouter'; // REGLA_008 interface UseSocketOptions { token: string; onTaskUpdate?: (data: TaskUpdatePayload) => void; onNotification?: (data: NotifPayload) => void; onPresence?: (data: PresencePayload) => void; } export function useSocket({ token, onTaskUpdate, onNotification, onPresence }: UseSocketOptions) { const socketRef = useRef(null); const [connected, setConnected] = useState(false); const [latency, setLatency] = useState(0); useEffect(() => { const socket = io(process.env.REACT_APP_WS_URL!, { auth: { token }, transports: ['websocket'], // no polling fallback reconnectionAttempts: 5, reconnectionDelay: 1500, // REGLA_008 backoff }); socketRef.current = socket; socket.on('connect', () => setConnected(true)); socket.on('disconnect', () => setConnected(false)); socket.on('task:update', onTaskUpdate ?? (() => {})); socket.on('notification:new', onNotification ?? (() => {})); socket.on('presence:join', onPresence ?? (() => {})); // Heartbeat para medir latencia const hbInterval = setInterval(() => { const t0 = Date.now(); socket.emit('heartbeat', { ts: t0 }); socket.once('heartbeat:ack', () => setLatency(Date.now() - t0)); }, 30000); return () => { clearInterval(hbInterval); socket.disconnect(); }; }, [token]); const emit = useCallback((event: string, data: unknown) => { if (socketRef.current?.connected) socketRef.current.emit(event, data); }, []); return { connected, latency, emit }; }
