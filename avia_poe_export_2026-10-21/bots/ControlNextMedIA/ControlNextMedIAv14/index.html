Habilitar KV v2 vault secrets enable -path=secret kv-v2 # 2. Habilitar AppRole vault auth enable approle # 3. Crear role backend con TTL vault write auth/approle/role/backend-role \ secret_id_ttl=10m \ token_num_uses=10 \ token_ttl=1h \ token_max_ttl=4h \ token_policies="backend-policy" # 4. Aplicar políticas vault policy write backend-policy policies/backend-policy.hcl vault policy write ci-policy policies/ci-policy.hcl # 5. Habilitar audit log → S3 vault audit enable file \ file_path=/vault/logs/audit.log echo "✅ Vault inicializado · ControlNextMedIA T3.5"
vault/policies/backend-policy.hcl
# ── Backend Policy · Least Privilege ── # Solo acceso a su propio path de entorno # Leer secretos del entorno activo path "secret/data/controlnextmedia/{{identity.entity.aliases.auth_approle_*.metadata.env}}/*" { capabilities = ["read", "list"] } # Renovar propio token path "auth/token/renew-self" { capabilities = ["update"] } # Leer propio rol AppRole (para secretId rotation) path "auth/approle/role/backend-role/secret-id" { capabilities = ["create", "update"] } # PROHIBIDO: acceso a otros entornos / admin paths path "sys/*" { capabilities = ["deny"] } path "auth/*" { capabilities = ["deny"] }
