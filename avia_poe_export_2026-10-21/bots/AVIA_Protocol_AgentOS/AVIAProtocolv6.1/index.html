🔐 Token Persistence Algorithm

// Multi-layer persistence with compression and encryption TokenPersistence.save(token) { // 1. Compress token data compressed = LZString.compress(token) // 2. Try storage methods in priority order strategies = [ () => localStorage.setItem('token', compressed), () => sessionStorage.setItem('token', compressed), () => setCookie('token', compressed, { httpOnly: false }), () => this.memoryStore.token = compressed ] // 3. Execute with fallback for (strategy of strategies) { try { strategy() return { success: true, method: strategy.name } } catch (e) { console.warn('Storage failed:', e) } } // 4. All failed - log error throw new Error('All storage methods failed') } // Token expiration check TokenPersistence.isExpired(token) { if (!token.expiresAt) return false return Date.now() > token.expiresAt }

🔄 Retry Logic with Exponential Backoff

// Intelligent retry with exponential backoff + jitter RetryLogic.scheduleReconnect(attempt) { // Base delay: 1s, 2s, 4s, 8s, 16s, 30s (max) baseDelay = Math.min(1000 * Math.pow(2, attempt), 30000) // Add jitter (±20%) to prevent thundering herd jitter = baseDelay * 0.2 * (Math.random() - 0.5) delay = baseDelay + jitter console.log(`Retry #${attempt} in ${delay}ms`) // Schedule reconnect this.reconnectTimer = setTimeout(() => { this.attemptReconnect() }, delay) // Update UI with countdown this.startCountdown(delay) } // Adaptive polling interval RetryLogic.adjustPollingInterval() { if (this.connectionStable) { // Stable: increase interval to reduce load this.pollingInterval = Math.min( this.pollingInterval * 1.5, 10000 // max 10s ) } else { // Unstable: decrease interval for faster recovery this.pollingInterval = Math.max( this.pollingInterval * 0.8, 1000 // min 1s ) } }

🛡️ Permission Middleware Chain

// Middleware chain execution MiddlewareChain.process(message, context) { // 1. Build middleware chain chain = [ AuthenticationMiddleware, PermissionMiddleware, RateLimitMiddleware, ValidationMiddleware, LoggingMiddleware ] // 2. Execute chain sequentially for (middleware of chain) { result = middleware.execute(message, context) if (!result.success) { // Middleware rejected - stop chain return { blocked: true, reason: result.reason, middleware: middleware.name } } // Middleware can modify context context = result.context } // 3. All middleware passed return { blocked: false, context } } // Permission middleware implementation PermissionMiddleware.execute(message, context) { requiredPermission = message.requiredPermission userPermissions = context.user?.permissions || [] if (!userPermissions.includes(requiredPermission)) { return { success: false, reason: `Missing permission: ${requiredPermission}` } } return { success: true, context } }
