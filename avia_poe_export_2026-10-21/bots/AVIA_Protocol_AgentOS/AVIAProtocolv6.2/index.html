🛡️ Solución al Error de Sandbox

⚠️
Error Original:
SecurityError: Failed to read 'localStorage' - The document is sandboxed and lacks the 'allow-same-origin' flag

🔍 ¿Por qué ocurre?

Los iframes sandboxed (como el que usa Poe para renderizar HTML) restringen el acceso a localStorage, sessionStorage y cookies por seguridad. Esto previene que scripts maliciosos accedan a datos sensibles.

✅ Solución: Storage Abstracto con Fallbacks

/** * 🛡️ StorageAdapter - Abstracción con fallbacks automáticos * Detecta el entorno y usa la mejor estrategia disponible */ var StorageAdapter = (function() { // ===== DETECCIÓN DE CAPACIDADES ===== var capabilities = { localStorage: false, sessionStorage: false, cookies: false, broadcastChannel: false }; // Test localStorage try { var test = '__storage_test__'; localStorage.setItem(test, test); localStorage.removeItem(test); capabilities.localStorage = true; } catch (e) { console.warn('[Storage] localStorage not available', e.name); } // Test sessionStorage try { var test = '__storage_test__'; sessionStorage.setItem(test, test); sessionStorage.removeItem(test); capabilities.sessionStorage = true; } catch (e) { console.warn('[Storage] sessionStorage not available', e.name); } // Test cookies try { document.cookie = '__test=1'; capabilities.cookies = document.cookie.indexOf('__test') !== -1; document.cookie = '__test=; max-age=0'; } catch (e) { console.warn('[Storage] Cookies not available', e.name); } // Test BroadcastChannel capabilities.broadcastChannel = typeof BroadcastChannel !== 'undefined'; console.log('[Storage] Capabilities detected:', capabilities); // ===== MEMORIA COMO FALLBACK UNIVERSAL ===== var memoryStore = {}; return { capabilities: capabilities, /** * 💾 Guardar dato con cascada de fallbacks */ setItem: function(key, value) { // Estrategia: localStorage → sessionStorage → memory if (capabilities.localStorage) { try { localStorage.setItem(key, value); console.log('[Storage] Saved to localStorage:', key); return true; } catch (e) { console.warn('[Storage] localStorage.setItem failed', e); } } if (capabilities.sessionStorage) { try { sessionStorage.setItem(key, value); console.log('[Storage] Saved to sessionStorage:', key); return true; } catch (e) { console.warn('[Storage] sessionStorage.setItem failed', e); } } // Fallback: memoria (no persiste entre recargas) memoryStore[key] = value; console.log('[Storage] Saved to memory:', key); return true; }, /** * 📥 Obtener dato */ getItem: function(key) { if (capabilities.localStorage) { try { return localStorage.getItem(key); } catch (e) {} } if (capabilities.sessionStorage) { try { return sessionStorage.getItem(key); } catch (e) {} } return memoryStore[key] || null; }, /** * 🗑️ Eliminar dato */ removeItem: function(key) { if (capabilities.localStorage) { try { localStorage.removeItem(key); } catch (e) {} } if (capabilities.sessionStorage) { try { sessionStorage.removeItem(key); } catch (e) {} } delete memoryStore[key]; }, /** * 🧹 Limpiar todo */ clear: function() { if (capabilities.localStorage) { try { localStorage.clear(); } catch (e) {} } if (capabilities.sessionStorage) { try { sessionStorage.clear(); } catch (e) {} } memoryStore = {}; } }; })();

🎯 Uso del Storage Adapter

// ANTES (vulnerable a sandbox) localStorage.setItem('token', myToken); // ❌ SecurityError // DESPUÉS (sandbox-safe) StorageAdapter.setItem('token', myToken); // ✅ Funciona siempre
✅ Ventajas

• Funciona en CUALQUIER entorno
• Auto-detección de capacidades
• Fallback transparente
• Zero configuration

⚠️ Limitaciones

• Memoria no persiste entre recargas
• Sin sincronización cross-tab en sandbox
• Requiere BroadcastChannel para sync

📡 Broadcast Channel API para Navegadores Antiguos

💡
Problema: Storage Events no funcionan en navegadores antiguos o sandboxes restrictivos. BroadcastChannel API ofrece una alternativa robusta con mejor soporte.

🔄 Comparación: Storage Events vs BroadcastChannel

Feature Storage Events BroadcastChannel
Soporte 95% (IE 9+) 85% (Chrome 54+, Firefox 38+)
Requiere Storage Sí (localStorage) No (memoria)
Funciona en Sandbox No Sí (con allow-scripts)
Dispara en misma tab No Sí
Payload máximo ~5-10MB (quota localStorage) Ilimitado (solo memoria)
Latencia ~1-5ms ~0.1-1ms

💻 Implementación Híbrida (Best of Both)

/** * 📡 HybridSync - Usa BroadcastChannel + Storage Events + Polling */ var HybridSync = { channels: {}, listeners: [], pollingInterval: null, lastState: {}, /** * 🚀 Inicializar sincronización híbrida */ init: function(channelName) { console.log('[HybridSync] Initializing for channel:', channelName); // MÉTODO 1: BroadcastChannel (preferido) if (StorageAdapter.capabilities.broadcastChannel) { try { this.channels[channelName] = new BroadcastChannel(channelName); this.channels[channelName].onmessage = function(event) { console.log('[HybridSync] 📡 BroadcastChannel message:', event.data); this.notifyListeners(event.data); }.bind(this); console.log('[HybridSync] ✅ BroadcastChannel active'); } catch (e) { console.warn('[HybridSync] BroadcastChannel failed', e); } } // MÉTODO 2: Storage Events (fallback) if (StorageAdapter.capabilities.localStorage) { window.addEventListener('storage', function(event) { if (!event.key || !event.key.startsWith(channelName)) return; console.log('[HybridSync] 💾 Storage event:', event.key); try { var data = JSON.parse(event.newValue || 'null'); this.notifyListeners(data); } catch (e) { console.warn('[HybridSync] Parse error', e); } }.bind(this)); console.log('[HybridSync] ✅ Storage Events active'); } // MÉTODO 3: Polling (último recurso para navegadores muy antiguos) this.startPolling(channelName); console.log('[HybridSync] 🟢 All sync methods initialized'); }, /** * 📤 Broadcast mensaje a todas las tabs */ broadcast: function(channelName, data) { console.log('[HybridSync] 📤 Broadcasting:', data); // Método 1: BroadcastChannel if (this.channels[channelName]) { try { this.channels[channelName].postMessage(data); console.log('[HybridSync] ✅ Sent via BroadcastChannel'); } catch (e) { console.warn('[HybridSync] BroadcastChannel send failed', e); } } // Método 2: Storage (para compatibilidad) if (StorageAdapter.capabilities.localStorage) { try { var key = channelName + '_sync_' + Date.now(); StorageAdapter.setItem(key, JSON.stringify(data)); // Auto-limpiar después de 5 segundos setTimeout(function() { StorageAdapter.removeItem(key); }, 5000); console.log('[HybridSync] ✅ Sent via Storage'); } catch (e) { console.warn('[HybridSync] Storage send failed', e); } } // Actualizar lastState para polling this.lastState[channelName] = data; }, /** * ⏱️ Polling como último recurso */ startPolling: function(channelName) { // Solo si no hay BroadcastChannel ni Storage Events if (StorageAdapter.capabilities.broadcastChannel || StorageAdapter.capabilities.localStorage) { console.log('[HybridSync] Polling not needed'); return; } console.log('[HybridSync] ⏱️ Starting polling (fallback)'); this.pollingInterval = setInterval(function() { // Verificar cambios en storage var currentState = StorageAdapter.getItem(channelName + '_state'); if (currentState !== this.lastState[channelName]) { console.log('[HybridSync] ⏱️ Change detected via polling'); try { var data = JSON.parse(currentState); this.lastState[channelName] = currentState; this.notifyListeners(data); } catch (e) {} } }.bind(this), 2000); // Check cada 2 segundos }, /** * 👂 Registrar listener */ onMessage: function(callback) { this.listeners.push(callback); }, /** * 🔔 Notificar listeners */ notifyListeners: function(data) { this.listeners.forEach(function(callback) { try { callback(data); } catch (e) { console.error('[HybridSync] Listener error', e); } }); }, /** * 🛑 Detener sincronización */ stop: function(channelName) { if (this.channels[channelName]) { this.channels[channelName].close(); delete this.channels[channelName]; } if (this.pollingInterval) { clearInterval(this.pollingInterval); this.pollingInterval = null; } console.log('[HybridSync] Stopped for channel:', channelName); } };

🎯 Ejemplo de Uso

// Inicializar sincronización HybridSync.init('token_sync'); // Escuchar mensajes HybridSync.onMessage(function(data) { console.log('Token change detected:', data); if (data.action === 'logout') { // Ejecutar logout local handleLogout(); } else if (data.action === 'token_refresh') { // Actualizar token local updateToken(data.token); } }); // Enviar mensaje a todas las tabs HybridSync.broadcast('token_sync', { action: 'logout', timestamp: Date.now() });

⏰ Estrategia Redis TTL para Token Expiration

⚡
Problema: La blacklist crece infinitamente si no se limpian los tokens expirados. Redis TTL es la solución óptima para auto-limpieza.

🏗️ Arquitectura Redis TTL

FLUJO DE INVALIDACIÓN CON REDIS TTL: 1. USER LOGOUT │ ├─► Decode JWT para obtener 'exp' (timestamp de expiración) │ ├─► Calcular TTL = exp - now (tiempo restante hasta expiración natural) │ ├─► Redis: SETEX blacklist:token TTL "1" │ └─► Token bloqueado hasta que expire naturalmente 2. TOKEN VERIFICATION │ ├─► Redis: GET blacklist:token │ ├─► Si existe → Token REVOCADO (401) │ └─► Si no existe → Continuar verificación JWT normal 3. AUTO-LIMPIEZA │ ├─► Redis elimina automáticamente cuando TTL = 0 │ └─► No necesita cron jobs ni limpieza manual BENEFICIOS: ✅ Zero memoria después de expiración ✅ Auto-limpieza sin cron jobs ✅ O(1) lookup performance ✅ Escala horizontalmente

💻 Implementación Servidor (Node.js)

const redis = require('redis'); const jwt = require('jsonwebtoken'); const redisClient = redis.createClient({ host: process.env.REDIS_HOST || 'localhost', port: process.env.REDIS_PORT || 6379, // Recomendado: usar conexión con retry retry_strategy: function(options) { if (options.total_retry_time > 1000 * 60 * 60) { return new Error('Redis retry timeout'); } return Math.min(options.attempt * 100, 3000); } }); /** * 🚪 Invalidar token con TTL inteligente */ async function invalidateToken(token) { try { // PASO 1: Decodificar JWT (sin verificar para obtener exp) const decoded = jwt.decode(token, { complete: true }); if (!decoded || !decoded.payload.exp) { throw new Error('Invalid token: no expiration'); } // PASO 2: Calcular tiempo restante hasta expiración const now = Math.floor(Date.now() / 1000); // Unix timestamp en segundos const exp = decoded.payload.exp; const ttl = exp - now; console.log('[Redis] Token TTL:', { exp: new Date(exp * 1000).toISOString(), now: new Date(now * 1000).toISOString(), ttl: ttl + 's', ttl_hours: (ttl / 3600).toFixed(2) + 'h' }); // PASO 3: Validar que el token no haya expirado ya if (ttl <= 0) { console.log('[Redis] ⚠️ Token already expired, no need to blacklist'); return { success: true, alreadyExpired: true }; } // PASO 4: Agregar a blacklist con TTL const key = `blacklist:${token}`; await redisClient.setEx( key, ttl, // TTL en segundos '1' // Valor dummy (solo nos importa la existencia) ); console.log('[Redis] ✅ Token blacklisted with TTL:', ttl + 's'); // PASO 5: (Opcional) Guardar metadata para debugging const metaKey = `blacklist:meta:${decoded.payload.jti || token.substr(0, 20)}`; await redisClient.setEx( metaKey, ttl, JSON.stringify({ userId: decoded.payload.sub, revokedAt: new Date().toISOString(), expiresAt: new Date(exp * 1000).toISOString(), reason: 'user_logout' }) ); return { success: true, ttl: ttl, expiresAt: new Date(exp * 1000).toISOString() }; } catch (error) { console.error('[Redis] ❌ Failed to invalidate token:', error); throw error; } } /** * 🛡️ Verificar si token está en blacklist */ async function isTokenBlacklisted(token) { try { const key = `blacklist:${token}`; const exists = await redisClient.get(key); if (exists) { console.log('[Redis] ❌ Token is blacklisted'); return true; } return false; } catch (error) { console.error('[Redis] ⚠️ Blacklist check failed (allow by default):', error); // IMPORTANTE: Si Redis falla, permitir acceso por defecto // Alternativa: bloquear todo si Redis crítico return false; } } /** * 📊 Estadísticas de blacklist */ async function getBlacklistStats() { try { // Contar tokens en blacklist const keys = await redisClient.keys('blacklist:*'); // Filtrar solo tokens (no metadata) const tokenKeys = keys.filter(k => !k.includes(':meta:')); // Obtener TTLs const ttls = await Promise.all( tokenKeys.slice(0, 100).map(k => redisClient.ttl(k)) ); const avgTTL = ttls.reduce((a, b) => a + b, 0) / ttls.length; return { total: tokenKeys.length, avgTTL: Math.round(avgTTL), avgTTLHours: (avgTTL / 3600).toFixed(2), memoryUsage: await redisClient.info('memory') }; } catch (error) { console.error('[Redis] Failed to get stats:', error); return null; } } // ===== ENDPOINTS ===== app.post('/api/auth/logout', async (req, res) => { try { const token = req.headers.authorization?.split(' ')[1]; if (!token) { return res.status(400).json({ error: 'No token provided' }); } // Invalidar con TTL inteligente const result = await invalidateToken(token); res.json({ success: true, message: 'Logout successful', ttl: result.ttl, expiresAt: result.expiresAt }); } catch (error) { console.error('[API] Logout error:', error); res.status(500).json({ error: 'Logout failed' }); } }); app.get('/api/admin/blacklist/stats', async (req, res) => { const stats = await getBlacklistStats(); res.json(stats); });

🎯 Adaptive Jitter para Alta Latencia

🌐
Problema: En redes con latencia alta y variable (ej: conexiones móviles 3G/4G, redes satelitales), el backoff fijo puede causar reconexiones prematuras o muy tardías.

📊 Análisis: Latencia Variable

ESCENARIOS DE LATENCIA: 1. DATACENTER (Latencia baja estable) ├─► Latencia media: 10-50ms ├─► Variación: ±5ms └─► Jitter óptimo: 10-20% 2. WIFI/4G (Latencia media variable) ├─► Latencia media: 50-200ms ├─► Variación: ±50ms └─► Jitter óptimo: 30-50% 3. 3G/SATELLITE (Latencia alta muy variable) ├─► Latencia media: 200-1000ms ├─► Variación: ±500ms └─► Jitter óptimo: 50-100% PROBLEMA CON JITTER FIJO: Jitter 20% con delay 4000ms: ├─► Rango: 3200ms - 4800ms (±800ms) └─► Si latencia real es ±2000ms → INSUFICIENTE SOLUCIÓN: ADAPTIVE JITTER ├─► Medir latencia de red en tiempo real ├─► Ajustar jitter_factor dinámicamente └─► jitter = max(base_jitter, latency_stddev * 2)

💻 Implementación: Adaptive Exponential Backoff

var AdaptiveBackoff = { // ===== CONFIGURACIÓN BASE ===== baseDelay: 1000, maxDelay: 60000, // Aumentado para alta latencia baseJitterFactor: 0.2, maxAttempts: 15, // ===== MÉTRICAS DE RED ===== latencyHistory: [], maxHistorySize: 50, lastPingTime: null, // ===== ESTADO ===== currentAttempt: 0, retryTimer: null, adaptiveJitterFactor: 0.2, /** * 📊 Medir latencia de red */ measureLatency: async function() { var start = performance.now(); try { // Ping al servidor (usar endpoint ligero) await fetch('/api/ping', { method: 'HEAD', cache: 'no-cache' }); var latency = performance.now() - start; // Agregar a historial this.latencyHistory.push(latency); // Mantener tamaño máximo if (this.latencyHistory.length > this.maxHistorySize) { this.latencyHistory.shift(); } console.log('[Adaptive] Latency measured:', Math.round(latency) + 'ms'); // Actualizar jitter factor this.updateJitterFactor(); return latency; } catch (error) { console.warn('[Adaptive] Latency measurement failed', error); return null; } }, /** * 🎯 Calcular estadísticas de latencia */ getLatencyStats: function() { if (this.latencyHistory.length === 0) { return null; } var sum = this.latencyHistory.reduce(function(a, b) { return a + b; }, 0); var mean = sum / this.latencyHistory.length; // Calcular desviación estándar var squaredDiffs = this.latencyHistory.map(function(val) { return Math.pow(val - mean, 2); }); var variance = squaredDiffs.reduce(function(a, b) { return a + b; }, 0) / this.latencyHistory.length; var stddev = Math.sqrt(variance); // Calcular percentiles var sorted = this.latencyHistory.slice().sort(function(a, b) { return a - b; }); var p50 = sorted[Math.floor(sorted.length * 0.5)]; var p95 = sorted[Math.floor(sorted.length * 0.95)]; var p99 = sorted[Math.floor(sorted.length * 0.99)]; return { mean: mean, stddev: stddev, min: Math.min.apply(null, this.latencyHistory), max: Math.max.apply(null, this.latencyHistory), p50: p50, p95: p95, p99: p99, samples: this.latencyHistory.length }; }, /** * 🔄 Actualizar jitter factor basado en latencia */ updateJitterFactor: function() { var stats = this.getLatencyStats(); if (!stats) { this.adaptiveJitterFactor = this.baseJitterFactor; return; } // Fórmula adaptiva: // jitter_factor = base + (stddev / mean) × multiplier // Esto da más jitter cuando hay más variabilidad var coefficientOfVariation = stats.stddev / stats.mean; var adaptiveFactor = this.baseJitterFactor + (coefficientOfVariation * 0.5); // Clamp entre 10% y 100% this.adaptiveJitterFactor = Math.max(0.1, Math.min(1.0, adaptiveFactor)); console.log('[Adaptive] Jitter factor updated:', { mean: Math.round(stats.mean) + 'ms', stddev: Math.round(stats.stddev) + 'ms', cv: coefficientOfVariation.toFixed(2), jitterFactor: (this.adaptiveJitterFactor * 100).toFixed(1) + '%' }); }, /** * 🎲 Calcular delay con jitter adaptivo */ calculateDelay: function(attempt) { // Exponential backoff base var exponentialDelay = this.baseDelay * Math.pow(2, attempt - 1); var cappedDelay = Math.min(exponentialDelay, this.maxDelay); // Usar jitter adaptivo (no fijo) var jitter = cappedDelay * this.adaptiveJitterFactor * (2 * Math.random() - 1); var finalDelay = Math.round(cappedDelay + jitter); // NUEVO: Añadir latencia esperada al delay var stats = this.getLatencyStats(); if (stats) { // Añadir 2× la latencia P95 para dar margen var latencyBuffer = stats.p95 * 2; finalDelay += latencyBuffer; console.log('[Adaptive] Added latency buffer:', Math.round(latencyBuffer) + 'ms'); } finalDelay = Math.max(finalDelay, 0); console.log('[Adaptive] Attempt ' + attempt + ':', { base: exponentialDelay + 'ms', jitter: Math.round(jitter) + 'ms', jitterFactor: (this.adaptiveJitterFactor * 100).toFixed(1) + '%', final: finalDelay + 'ms' }); return finalDelay; }, /** * 🔄 Programar retry con medición de latencia */ scheduleRetry: async function(callback) { if (this.currentAttempt >= this.maxAttempts) { console.error('[Adaptive] Max attempts reached'); return false; } this.currentAttempt++; // Medir latencia antes de calcular delay await this.measureLatency(); // Calcular delay con jitter adaptivo var delay = this.calculateDelay(this.currentAttempt); console.log('[Adaptive] Scheduling retry #' + this.currentAttempt + ' in ' + delay + 'ms'); if (this.retryTimer) { clearTimeout(this.retryTimer); } this.retryTimer = setTimeout(function() { console.log('[Adaptive] Executing retry #' + this.currentAttempt); callback(); }.bind(this), delay); return { attempt: this.currentAttempt, delay: delay }; }, /** * ✅ Reset */ reset: function() { console.log('[Adaptive] Reset'); this.currentAttempt = 0; if (this.retryTimer) { clearTimeout(this.retryTimer); this.retryTimer = null; } // NO resetear historial de latencia (mantener aprendizaje) } };
